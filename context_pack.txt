```js
document.addEventListener('DOMContentLoaded', () => {
  // 读取所有存储的数据
  chrome.storage.local.get(null, (items) => {
    const listDiv = document.getElementById('stats-list');
    
    // 将对象转为数组并排序: [[word, count], ...]
    const sortedWords = Object.entries(items)
      .sort((a, b) => b[1] - a[1]); // 按次数降序排列

    if (sortedWords.length === 0) {
      listDiv.innerHTML = "<p>Go hover over some words!</p>";
      return;
    }

    let html = '<ul>';
    sortedWords.forEach(([word, count]) => {
      html += `
        <li>
          <span class="word">${word}</span>
          <span class="count">${count} times</span>
        </li>
      `;
    });
    html += '</ul>';
    
    listDiv.innerHTML = html;
  });
});
```

```js
chrome.runtime.onInstalled.addListener(() => {
  // 1. 先移除已有的菜单，防止重复ID报错
  chrome.contextMenus.removeAll(() => {
    // 2. 只有移除成功后，才创建新的

      id: "add-tech-word",
      title: "Add \"%s\" to Tech Vocabulary",
      contexts: ["selection"]
    });
  });
});

chrome.contextMenus.onClicked.addListener((info, tab) => {
  if (info.menuItemId === "add-tech-word" && info.selectionText) {
    chrome.tabs.sendMessage(tab.id, {
      action: "prompt_for_definition",
      word: info.selectionText.trim()
    });
  }
});
```

--- FILE: content.js ---
```js
let vocabulary = {};
let masteredWords = new Set();
const vocabUrl = chrome.runtime.getURL('vocabulary.json');
let scanTimeout = null;

initialize();

function initialize() {
  Promise.all([
    chrome.storage.local.get(['mastered_list', 'custom_vocab']),
    fetch(vocabUrl).then(res => res.json())
  ]).then(([storageData, baseVocab]) => {
    if (storageData.mastered_list) {
      masteredWords = new Set(storageData.mastered_list);
    }
    const customVocab = storageData.custom_vocab || {};
    vocabulary = { ...baseVocab, ...customVocab };

    scanAndHighlight(document.body);
    observeChanges();
  });
}

chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === "prompt_for_definition") {
    handleAddWord(request.word);
  }
});

function handleAddWord(rawWord) {
  const wordKey = rawWord.toLowerCase();
  const definition = prompt(`添加生词:\n"${rawWord}"\n\n请输入中文释义:`, "自定义笔记");
  
  if (definition && definition.trim() !== "") {
    vocabulary[wordKey] = definition;
    chrome.storage.local.get(['custom_vocab'], (result) => {
      const currentCustom = result.custom_vocab || {};
      currentCustom[wordKey] = definition;
      chrome.storage.local.set({ custom_vocab: currentCustom }, () => {
        if (masteredWords.has(wordKey)) {
          masteredWords.delete(wordKey);
          chrome.storage.local.set({ mastered_list: Array.from(masteredWords) });
        }
        scanAndHighlight(document.body);
      });
    });
  }
}

function scanAndHighlight(root) {
  const words = Object.keys(vocabulary);
  if (words.length === 0) return;
  
  const activeWords = words.filter(w => !masteredWords.has(w));
  if (activeWords.length === 0) return;

  const sortedWords = activeWords.sort((a, b) => b.length - a.length);
  const pattern = `\\b(${sortedWords.join('|')})(?:s|es|d|ed|ing|ly)?\\b`;
  const regex = new RegExp(pattern, 'gi'); 

  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null, false);
  let node;
  const nodesToReplace = [];

  while (node = walker.nextNode()) {
    const parent = node.parentNode;
    if (parent.nodeName === 'SCRIPT' || 
        parent.nodeName === 'STYLE' || 
        parent.classList.contains('tech-word-highlight') ||
        parent.isContentEditable) { 
      continue;
    }
    if (regex.test(node.nodeValue)) {
      nodesToReplace.push(node);
    }
  }

  nodesToReplace.forEach(node => {
    const fragment = document.createDocumentFragment();
    const text = node.nodeValue;
    let lastIndex = 0;
    
    text.replace(regex, (match, rootWord, offset) => {
      const before = text.slice(lastIndex, offset);
      if (before) fragment.appendChild(document.createTextNode(before));
      
      const lowerMatch = match.toLowerCase();
      const dictionaryKey = findRootInVocab(lowerMatch);

      if (dictionaryKey && !masteredWords.has(dictionaryKey)) {
        const span = document.createElement('span');
        span.className = 'tech-word-highlight';
        span.textContent = match;
        span.dataset.word = dictionaryKey;
        
        // --- 极简修改：初始只存定义，不显示任何提示文案 ---
        span.dataset.tooltip = vocabulary[dictionaryKey];
        
        span.addEventListener('mouseenter', () => handleHover(dictionaryKey, span));
        span.addEventListener('click', (e) => {
          e.stopPropagation();
          e.preventDefault();
          if (e.altKey) {
            markAsMastered(dictionaryKey);
          } else {
            speakText(match);
            // 这里的 Toast 提示也可以去掉了，既然你自己知道
          }
        });
        fragment.appendChild(span);
      } else {
        fragment.appendChild(document.createTextNode(match));
      }
      lastIndex = offset + match.length;
      return match;
    });

    const remaining = text.slice(lastIndex);
    if (remaining) fragment.appendChild(document.createTextNode(remaining));
    node.parentNode.replaceChild(fragment, node);
  });
}

function findRootInVocab(word) {
  if (vocabulary[word]) return word;
  const suffixes = ['ing', 'ed', 'es', 's', 'd', 'ly'];
  for (const suffix of suffixes) {
    if (word.endsWith(suffix)) {
      const stem = word.slice(0, -suffix.length);
      if (vocabulary[stem]) return stem;
    }
  }
  return null;
}

function handleHover(word, element) {
  chrome.storage.local.get([word], (result) => {
    const newCount = (result[word] || 0) + 1;
    chrome.storage.local.set({ [word]: newCount });
    const def = vocabulary[word];
    
    // --- 极简修改：悬停时显示 "定义 + 换行 + [N]" ---
    if (element) {
        element.dataset.tooltip = `${def}\n[${newCount}]`;
    }
  });
}

function speakText(text) {
  window.speechSynthesis.cancel();
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = 'en-US';
  window.speechSynthesis.speak(utterance);
}

function markAsMastered(word) {
  // 保持 confirm 防止误触，但文案简化
  if (confirm(`标记为已掌握：${word}?`)) {
    masteredWords.add(word);
    chrome.storage.local.set({ 'mastered_list': Array.from(masteredWords) });
    const highlights = document.querySelectorAll(`.tech-word-highlight[data-word="${word}"]`);
    highlights.forEach(el => {
      const text = document.createTextNode(el.textContent);
      el.parentNode.replaceChild(text, el);
    });
  }
}

function observeChanges() {
  const observer = new MutationObserver((mutations) => {
    if (scanTimeout) clearTimeout(scanTimeout);
    scanTimeout = setTimeout(() => {
      let hasNewElements = false;
      for (const mutation of mutations) {
        if (mutation.addedNodes.length > 0) {
          hasNewElements = true;
          break;
        }
      }
      if (hasNewElements) {
        scanAndHighlight(document.body);
      }
    }, 1000);
  });
  observer.observe(document.body, { childList: true, subtree: true });
}
// 1. 初始化：在页面底部插入那个全局 Tooltip
const tooltipEl = document.createElement('div');
tooltipEl.id = 'neural-tooltip-container';
document.body.appendChild(tooltipEl);

function handleHover(event) {
    const target = event.target;
    // 检查是否是我们标记的单词
    if (!target.classList.contains('tech-word-highlight')) return;

    // 获取字典定义 key
    const dictionaryKey = target.dataset.key; 
    if (!dictionaryKey) return;

    // === 核心逻辑：计算位置 ===
    // getBoundingClientRect() 获取元素相对于“视口”的坐标
    const rect = target.getBoundingClientRect();

    // 更新 Tooltip 内容
    // (假设 vocabulary 是你之前的字典对象)
    const def = vocabulary[dictionaryKey] || "Unknown";
    tooltipEl.innerText = def;

    // 移动 Tooltip
    // 让它显示在单词下方 5px 的位置
    // rect.left = 单词左边缘, rect.bottom = 单词下边缘
    tooltipEl.style.left = `${rect.left}px`;
    tooltipEl.style.top = `${rect.bottom + 5}px`;

    // 显示
    tooltipEl.classList.add('active');
}

function handleLeave() {
    // 隐藏
    tooltipEl.classList.remove('active');
}

// 2. 绑定事件 (使用事件委托，性能更好)
// 只要监听 document 即可，不用给每个单词都绑事件
document.addEventListener('mouseover', handleHover);
document.addEventListener('mouseout', handleLeave);
```
